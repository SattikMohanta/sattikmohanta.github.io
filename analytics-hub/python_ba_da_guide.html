Outlier Detection DATA CLEAN 
# — Z-score method —
from scipy import stats
z_scores = stats.zscore(df['col'].dropna())
outliers = df[abs(z_scores) > 3]           # beyond 3σ

# — IQR method (robust, non-parametric) —
Q1 = df['col'].quantile(0.25)

IQR = Q3 - Q1
lower = Q1 - 1.5 * IQR
upper = Q3 + 1.5 * IQR

p99 = df['col'].quantile(0.99)
df['col'] = df['col'].clip(p01, p99)

Window Functions & Running Calculations

🎯 Interview Differentiator
Window functions are where most BA/DA candidates stumble in interviews. Mastering transform(), shift(), and rolling() puts you in the top 10% of candidates.
All Window Function Patterns WINDOW FNS 
# — ROW_NUMBER() per group —
df['row_num'] = df.groupby('dept').cumcount() + 1

# — RANK() per group (handles ties) —
df['rank'] = df.groupby('dept')['salary'].rank(method='min', ascending=False)
# rank methods: 'average' | 'min' (RANK) | 'max' | 'first' (ROW_NUMBER) | 'dense' (DENSE_RANK)

# — DENSE_RANK() —
df['dense_rank'] = df.groupby('dept')['salary'].rank(method='dense', ascending=False)

# — LAG / LEAD —
df['prev_month_rev'] = df.groupby('product')['revenue'].shift(1)  # LAG(1)
df['next_month_rev'] = df.groupby('product')['revenue'].shift(-1) # LEAD(1)

# — MoM % change —
df['mom_growth'] = df.groupby('product')['revenue'].pct_change() * 100

# — Running total (CUMSUM) per group —
df['running_total'] = df.groupby('region')['revenue'].cumsum()

# — % of group total (SUM OVER PARTITION BY) —
df['group_total'] = df.groupby('region')['revenue'].transform('sum')
df['pct_of_group'] = df['revenue'] / df['group_total'] * 100

# — Rolling averages (moving windows) —
df['ma_3'] = df['revenue'].rolling(window=3, min_periods=1).mean()  # 3-period MA
df['ma_7'] = df['revenue'].rolling(7).mean()              # 7-day MA
df['roll_std'] = df['revenue'].rolling(30).std()          # rolling volatility

# — Expanding window (from start to current row) —
df['ytd_total'] = df['revenue'].expanding().sum()

# — Exponential weighted mean (trend-aware) —
df['ewm_revenue'] = df['revenue'].ewm(span=7, adjust=False).mean()

# — Top N per group (like ROW_NUMBER + filter) —
top2_per_region = (
    df.sort_values(['region', 'revenue'], ascending=[True, False])
      .groupby('region')
      .head(2)
)

Method Chaining (Fluent API) INTERVIEW GOLD 
# Instead of this (amateur approach):
df1 = df.dropna()
df2 = df1[df1['revenue'] > 0]
df3 = df2.groupby('region')['revenue'].sum()

# Do this (professional method chaining):
result = (
    df
    .dropna(subset=['revenue', 'region'])
    .query('revenue > 0 and region != "UNKNOWN"')
    .assign(
        revenue_k = lambda x: x['revenue'] / 1000,
        month = lambda x: x['date'].dt.to_period('M')
    )
    .groupby(['region', 'month'])
    .agg(total_rev=('revenue_k', 'sum'),
         order_ct  =('order_id', 'count'))
    .reset_index()
    .sort_values('total_rev', ascending=False)
)
# ↑ readable, debuggable, no temp vars
Performance Optimization INTERVIEW GOLD 
# — Vectorize! Never loop over rows —
# BAD (slow):
for i, row in df.iterrows():          # 1000x slower!
    df.at[i, 'total'] = row['qty'] * row['price']

# GOOD (vectorized):
df['total'] = df['qty'] * df['price']

# — Use categoricals for repeated strings —
df['region'] = df['region'].astype('category')  # 80% memory reduction

# — Downsample numeric types —
df['val'] = pd.to_numeric(df['val'], downcast='float')  # float64 → float32

# — query() is faster on large DataFrames —
df.query('revenue > @threshold and region == @target_region')

# — Check memory usage —
df.info(memory_usage='deep')
df.memory_usage(deep=True).sum() / 1e6  # MB

# When SQL is genuinely easier — use it directly!
import pandasql as psql

orders = pd.read_csv('orders.csv')
customers = pd.read_csv('customers.csv')

result = psql.sqldf("""
    SELECT 
        c.region,
        COUNT(DISTINCT o.customer_id)   AS unique_customers,
        SUM(o.revenue)                  AS total_revenue,
        AVG(o.revenue)                  AS avg_order_value,
        RANK() OVER (
            ORDER BY SUM(o.revenue) DESC
        ) AS revenue_rank
    FROM orders o
    JOIN customers c ON o.customer_id = c.id
    WHERE o.status = 'completed'
    GROUP BY c.region
    HAVING COUNT(*) > 100
    ORDER BY total_revenue DESC
""", locals())
# ↑ Instantly impresses SQL-first interviewers!
Cohort & Retention Analysis BA SPECIFIC 
# — Cohort retention matrix —
df['cohort_month'] = df.groupby('user_id')['date'].\
    transform('min').dt.to_period('M')
df['order_month'] = df['date'].dt.to_period('M')
df['period'] = (df['order_month'] - df['cohort_month']).apply(lambda x: x.n)

cohort_size = df.groupby('cohort_month')['user_id'].nunique()
retention_matrix = (
    df.groupby(['cohort_month', 'period'])['user_id']
      .nunique()
      .unstack()
      .divide(cohort_size, axis=0)
)

# — RFM Segmentation —
snapshot = df['date'].max()
rfm = df.groupby('customer_id').agg(
    Recency   = ('date',     lambda x: (snapshot - x.max()).days),
    Frequency = ('order_id', 'count'),
    Monetary  = ('revenue',  'sum')
).reset_index()
rfm['R_score'] = pd.qcut(rfm['Recency'],   5, labels=[5,4,3,2,1])
rfm['F_score'] = pd.qcut(rfm['Frequency'], 5, labels=[1,2,3,4,5])

Automation & Scheduling Patterns ETL PIPELINE 
# — Production ETL function template —
import logging
logging.basicConfig(level=logging.INFO,
    format='%(asctime)s %(levelname)s %(message)s')

def run_etl(start_date: str, end_date: str) -> dict:
    """Incremental ETL with logging & error handling."""
    stats = {'extracted': 0, 'loaded': 0, 'errors': 0}
    try:
        logging.info(f"Extracting {start_date} → {end_date}")
        df = extract(start_date, end_date)
        stats['extracted'] = len(df)
        
        df = transform(df)
        validate(df)                         # assert checks
        
        rows = load(df)
        stats['loaded'] = rows
        logging.info(f"✓ Loaded {rows} rows successfully")
        
    except Exception as e:
        stats['errors'] += 1
        logging.error(f"ETL failed: {e}")
        raise
    
    return stats
